# Optional build-time dependencies for tools/embed_html.py.
#
# The dashboard is minified before it is gzip-embedded into the firmware.
# JS: terser (preferred — `cd tools && npm install`, see tools/package.json),
# falling back to rjsmin (comment/whitespace only). CSS: rcssmin. Compression:
# zopfli emits a gzip-compatible stream a few % smaller than gzip -9.
# embed_html.py reports what it used on its status line; without ANY JS
# minifier the embed is ~100 KB gz larger (≈5 % of the OTA slot). Install:
#
#   pip3 install -r tools/requirements.txt
#
rjsmin==1.2.5
rcssmin==1.2.2
zopfli
